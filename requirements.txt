# The harness and its tests. Pinned: the corpus in this repository was verified
# byte-identical against a fresh reading with exactly these versions
# (`python -m harness drift`: 289 artefacts compared, 0 divergent).
pyyaml==6.0.3
pytest==8.4.2

# `harness/ledger.py` reads the PDFs through PyMuPDF: `get_text("dict")` gives
# positioned spans with their font size, which is what the span ledger is made
# of. It is imported inside the reader, so everything except `harness ledger`
# works without it.
pymupdf==1.26.5

# `harness/oasis.py` validates every committed AKN document against the schema
# vendored under `vendor/akn/`. lxml is the only library that enforces an XSD,
# and it is imported inside the validator - so everything except `harness oasis`
# works without it, the way everything except `harness ledger` works without
# PyMuPDF. It is pinned for the same reason the others are: the conformance
# `fixtures/oasis-conformance.yaml` declares was read with exactly this reader,
# and a schema engine that reports different messages would move a declaration
# nobody changed.
lxml==6.0.2
